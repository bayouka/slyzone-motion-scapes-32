function renderProject(id, tab='overview') {
  const project = state.projects.find(p => p.id === id);
  if (!project) return empty('Projet introuvable','Vous n’avez peut-être plus accès à ce projet.');
  const actions = state.actions.filter(a=>a.project_id===id);
  const milestones = state.projectCache.get(id)?.milestones || [];
  if (!state.projectCache.has(id)) loadProject(id);
  const done = actions.filter(a=>a.status==='done').length;
  const progress = actions.length ? Math.round(done/actions.length*100) : 0;
  const tabs = [['overview','Vue d’ensemble'],['roadmap','Roadmap'],['work','Travail'],['messages','Messages'],['decisions','Décisions'],['meetings','Réunions'],['resources','Ressources']];
  const href = (t)=>`#/projects/${id}/${t}`;
  const tabHtml = `<div class="tabs">${tabs.map(([key,label])=>`<a href="${href(key)}" class="${tab===key?'active':''}">${label}</a>`).join('')}</div>`;
  const canManageProject = ['owner','admin'].includes(state.workspaceRole);
  const head = `<div class="section-head"><div><span class="eyebrow">Projet</span><h2>${esc(project.name)}</h2><p>${esc(project.objective || 'Aucun objectif renseigné.')}</p></div><div class="head-actions"><span class="pill ${healthTone(project.health)}"><span class="dot"></span>${healthLabel(project.health)}</span><button class="btn small" data-action="edit-project" data-project="${project.id}">Modifier</button>${canManageProject?`<button class="btn small" data-action="manage-project" data-project="${project.id}">Gérer</button>`:''}</div></div>${tabHtml}`;
  if (tab==='roadmap') return head + projectRoadmap(project, actions, milestones);
  if (tab==='work') return head + projectWork(project, actions, milestones);
  if (tab==='messages') return head + projectMessages(project);
  if (tab==='decisions') return head + projectDecisions(project);
  if (tab==='meetings') return head + projectMeetings(project);
  if (tab==='resources') return head + projectResources(project);
  return head + projectOverview(project, actions, milestones, progress);
}
